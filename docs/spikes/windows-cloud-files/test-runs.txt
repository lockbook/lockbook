$ErrorActionPreference = 'Stop'
$PA = 'C:\Users\travis\Lockbook-alice\Team\plan.md'; $PB = 'C:\Users\travis\Lockbook-bob\Team\plan.md'
function L($m) { Log 't3.ps1' $m }
function ReadText($p) { [IO.File]::ReadAllText($p) }
try {
[IO.File]::WriteAllText($PA, "# Team plan`r`n`r`n- ship the Windows spike`r`n- alice: edited from Windows`r`n- bob: edited from Windows`r`n")
Start-Sleep 9
# 1. simultaneous: both editors start from the same text, both save before either syncs
$base = ReadText $PA; $null = ReadText $PB
L "base: $base"
$ta = $base.Replace('# Team plan', '# Team plan (alice retitled)')
$tb = $base + "- bob: added at the same time`r`n"
[IO.File]::WriteAllText($PA, $ta); [IO.File]::WriteAllText($PB, $tb)
Start-Sleep 14
L "simultaneous -> alice: $(ReadText $PA)"
L "simultaneous -> bob: $(ReadText $PB)"
# 2. stale editor: bob's buffer predates alice's save, which lands on bob's disk first
$stale = ReadText $PB
[IO.File]::WriteAllText($PA, (ReadText $PA) + "- alice: second edit`r`n")
Start-Sleep 9
L "bob disk before his save: $(ReadText $PB)"
[IO.File]::WriteAllText($PB, $stale.Replace('ship the Windows spike', 'ship the Windows spike (bob, stale buffer)'))
Start-Sleep 12
L "stale -> alice: $(ReadText $PA)"
L "stale -> bob: $(ReadText $PB)"
} catch { L "ERR $($_.Exception.Message) at $($_.InvocationInfo.ScriptLineNumber)" }
L '--- alice log'; Get-Content C:\Users\travis\cf-alice.log -Tail 10 | ForEach-Object { L $_ }
